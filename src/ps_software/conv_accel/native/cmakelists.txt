cmake_minimum_required(VERSION 3.15)
project(y26_conv_connection LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
find_package(ncnn CONFIG REQUIRED)
add_library(y26_first STATIC first_layer.cpp tile_transport.cpp fusion.cpp)
target_include_directories(y26_first PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(y26_first PUBLIC ncnn)
target_compile_options(y26_first PRIVATE -Wall -Wextra -ffp-contract=off)
add_executable(y26-conv-check conv_check.cpp)
target_link_libraries(y26-conv-check PRIVATE y26_first)
add_executable(tile_transport_test tests/tile_transport_test.cpp)
target_link_libraries(tile_transport_test PRIVATE y26_first)
add_executable(first_layer_test tests/first_layer_test.cpp)
target_link_libraries(first_layer_test PRIVATE y26_first)
add_executable(fusion_test tests/fusion_test.cpp)
target_link_libraries(fusion_test PRIVATE y26_first)
install(TARGETS y26-conv-check RUNTIME DESTINATION bin)
add_library(y26_chain STATIC two_layer.cpp channel_conv.cpp)
target_include_directories(y26_chain PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(y26_chain PUBLIC y26_first)
target_compile_options(y26_chain PRIVATE -Wall -Wextra -ffp-contract=off)
add_executable(y26-chain-check chain_check.cpp)
target_link_libraries(y26-chain-check PRIVATE y26_chain)
foreach(name dma_limits channel_conv two_layer_graph)
    add_executable(${name}_test tests/${name}_test.cpp)
    target_link_libraries(${name}_test PRIVATE y26_chain)
endforeach()
install(TARGETS y26-chain-check RUNTIME DESTINATION bin)
add_executable(y26-dma-stream dma_stream_check.cpp)
target_link_libraries(y26-dma-stream PRIVATE y26_first)
install(TARGETS y26-dma-stream RUNTIME DESTINATION bin)
