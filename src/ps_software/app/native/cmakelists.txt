cmake_minimum_required(VERSION 3.15)
project(zynq_ps_inference LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(ncnn CONFIG REQUIRED)
add_executable(yolo26_detect yolo26_detect.cpp)
set(ACCEL_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../../conv_accel/native)
target_sources(yolo26_detect PRIVATE ${ACCEL_DIR}/raw_frame.cpp ${ACCEL_DIR}/two_layer.cpp ${ACCEL_DIR}/channel_conv.cpp ${ACCEL_DIR}/fusion.cpp ${ACCEL_DIR}/tile_transport.cpp ${ACCEL_DIR}/resident_prefix.cpp)
target_compile_definitions(yolo26_detect PRIVATE _FILE_OFFSET_BITS=64)
target_include_directories(yolo26_detect PRIVATE third_party/stb ${ACCEL_DIR})
target_compile_options(yolo26_detect PRIVATE -ffp-contract=off)
target_link_libraries(yolo26_detect PRIVATE ncnn)
target_compile_options(yolo26_detect PRIVATE -Wall -Wextra)
install(TARGETS yolo26_detect RUNTIME DESTINATION bin)

include(CTest)
if(BUILD_TESTING)
    add_executable(detection_custom_test tests/detection_custom_test.cpp)
    target_link_libraries(detection_custom_test PRIVATE ncnn)
    add_test(NAME detection_postprocess COMMAND detection_custom_test)
endif()
